What is a REST API?

REST (Representational State Transfer) is a set of rules for how web services communicate using HTTP. The key rule for beginners: URLs describe things (nouns), not actions (verbs).

❌ Wrong — action URLs
/getStudents
/createStudent
/updateStudent/5
/deleteStudent/5
✅ Correct — resource URLs
GET /students
POST /students
PUT /students/5
DELETE /students/5
The URL says what. The HTTP method says what to do with it. This is REST's core idea — clean, predictable, and universally understood.

Full Controller — All 5 CRUD Functions

backend/controllers/studentController.jsJS
const Student = require('../models/Student');

// ── Shared error handler (DRY principle) ──────────
const handleError = (res, err) => {
  if (err.name === 'ValidationError') {
    const msgs = Object.values(err.errors).map(e => e.message);
    return res.status(400).json({ success: false, errors: msgs });
  }
  if (err.code === 11000)
    return res.status(409).json({ success: false, message: 'Email already exists' });
  if (err.name === 'CastError')
    return res.status(400).json({ success: false, message: 'Invalid ID format' });
  res.status(500).json({ success: false, message: err.message });
};

// ── 1. GET all students ───────────────────────────
const getAllStudents = async (req, res) => {
  try {
    const students = await Student.find().sort({ createdAt: -1 });
    res.json({ success: true, count: students.length, data: students });
  } catch (err) { handleError(res, err); }
};

// ── 2. GET one student by ID ──────────────────────
const getStudentById = async (req, res) => {
  try {
    const student = await Student.findById(req.params.id);
    if (!student)
      return res.status(404).json({ success: false, message: 'Student not found' });
    res.json({ success: true, data: student });
  } catch (err) { handleError(res, err); }
};

// ── 3. CREATE a new student ───────────────────────
const createStudent = async (req, res) => {
  try {
    const student = await Student.create(req.body);
    res.status(201).json({
      success: true,
      message: 'Student created successfully',
      data: student,
    });
  } catch (err) { handleError(res, err); }
};

// ── 4. UPDATE a student ───────────────────────────
const updateStudent = async (req, res) => {
  try {
    const student = await Student.findByIdAndUpdate(
      req.params.id,
      req.body,
      { new: true, runValidators: true }
    );
    if (!student)
      return res.status(404).json({ success: false, message: 'Student not found' });
    res.json({ success: true, message: 'Updated successfully', data: student });
  } catch (err) { handleError(res, err); }
};

// ── 5. DELETE a student ───────────────────────────
const deleteStudent = async (req, res) => {
  try {
    const student = await Student.findByIdAndDelete(req.params.id);
    if (!student)
      return res.status(404).json({ success: false, message: 'Student not found' });
    res.json({ success: true, message: 'Student deleted successfully', data: {} });
  } catch (err) { handleError(res, err); }
};

module.exports = { getAllStudents, getStudentById, createStudent, updateStudent, deleteStudent };

Routes File — Final

backend/routes/studentRoutes.jsJS
const express = require('express');
const router  = express.Router();
const {
  getAllStudents, getStudentById,
  createStudent, updateStudent, deleteStudent,
} = require('../controllers/studentController');

router.route('/')   .get(getAllStudents).post(createStudent);
router.route('/:id').get(getStudentById).put(updateStudent).delete(deleteStudent);

module.exports = router;

Full Jest + Supertest Test Suite (11 Tests)

backend/tests/student.test.jsJS
const request  = require('supertest');
const mongoose = require('mongoose');
const app      = require('../server');
const Student  = require('../models/Student');

beforeAll(async () => {
  await mongoose.connect(
    process.env.MONGO_URI_TEST || 'mongodb://localhost:27017/studentTestDB'
  );
});
afterEach (async () => { await Student.deleteMany({}); });
afterAll  (async () => { await mongoose.connection.close(); });

const seed = () => Student.create({
  name: 'Kaveesha', email: 'k@test.com', age: 21, course: 'IT',
});

// ── POST ─────────────────────────────────────────
describe('POST /api/students', () => {
  it('creates student → 201', async () => {
    const res = await request(app).post('/api/students')
      .send({ name: 'Nuwan', email: 'n@test.com', age: 22, course: 'IT' });
    expect(res.statusCode).toBe(201);
    expect(res.body.data._id).toBeDefined();
  });
  it('missing name → 400', async () => {
    const res = await request(app).post('/api/students')
      .send({ email: 'x@t.com', course: 'IT' });
    expect(res.statusCode).toBe(400);
  });
  it('duplicate email → 409', async () => {
    await seed();
    const res = await request(app).post('/api/students')
      .send({ name: 'B', email: 'k@test.com', course: 'IT' });
    expect(res.statusCode).toBe(409);
  });
});

// ── GET ALL ───────────────────────────────────────
describe('GET /api/students', () => {
  it('returns all students → 200', async () => {
    await seed();
    const res = await request(app).get('/api/students');
    expect(res.statusCode).toBe(200);
    expect(res.body.count).toBe(1);
  });
  it('empty DB → count 0', async () => {
    const res = await request(app).get('/api/students');
    expect(res.body.count).toBe(0);
  });
});

// ── GET ONE ───────────────────────────────────────
describe('GET /api/students/:id', () => {
  it('returns student by id → 200', async () => {
    const s   = await seed();
    const res = await request(app).get(`/api/students/${s._id}`);
    expect(res.statusCode).toBe(200);
    expect(res.body.data.name).toBe('Kaveesha');
  });
  it('wrong id → 404', async () => {
    const res = await request(app).get('/api/students/64f000000000000000000000');
    expect(res.statusCode).toBe(404);
  });
});

// ── PUT ───────────────────────────────────────────
describe('PUT /api/students/:id', () => {
  it('updates student → 200', async () => {
    const s   = await seed();
    const res = await request(app).put(`/api/students/${s._id}`).send({ age: 25 });
    expect(res.statusCode).toBe(200);
    expect(res.body.data.age).toBe(25);
  });
  it('wrong id → 404', async () => {
    const res = await request(app).put('/api/students/64f000000000000000000000').send({ age: 30 });
    expect(res.statusCode).toBe(404);
  });
});

// ── DELETE ────────────────────────────────────────
describe('DELETE /api/students/:id', () => {
  it('deletes student → 200', async () => {
    const s   = await seed();
    const res = await request(app).delete(`/api/students/${s._id}`);
    expect(res.statusCode).toBe(200);
    const check = await Student.findById(s._id);
    expect(check).toBeNull(); // truly gone from DB
  });
  it('wrong id → 404', async () => {
    const res = await request(app).delete('/api/students/64f000000000000000000000');
    expect(res.statusCode).toBe(404);
  });
});
